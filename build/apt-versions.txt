# The exact Debian package versions both tiers install, read from the pinned snapshot in
# build/snapshot.env (trixie @ 20260901T000000Z, plus trixie-security at the same instant).
#
# docs/test-harness.md requires the authoritative tier to install *the exact versions the
# ISO installs*, so a skew fails in CI rather than on the appliance. This file is the one
# pinned list, and it has three consumers: build/Dockerfile.test installs BOTH groups,
# build/mkiso.sh installs the appliance group ONLY, and build/verify.py parses it to
# assert that no harness package reached the rootfs.
#
# The `# @group` markers are MACHINE-READABLE AND LOAD-BEARING. A package outside any
# group is a parse error in verify.py, never a guess. In the Alpine predecessor the split
# was once a prose comment with `py3-pip` on the far side of it, and a naive parser — as
# Dockerfile.test is, and is right to be, because it wants both halves — would have put a
# package manager in the rootfs and defeated a published claim with nothing noticing.
#
# These are TOP-LEVEL PINS, not the closure. build/fetch-inputs.sh (M2) expands them with
# apt's own resolver at this snapshot and writes every resolved .deb into
# build/inputs.sha256, which is what the build actually gates on.
#
# NO PYTHON PACKAGE IS PINNED HERE except the interpreter itself and pip. Every Python
# dependency of the appliance and of the harness comes from pyproject.toml, resolved in
# uv.lock and listed in build/wheel-versions.txt. Adding a `python3-*` line below is a
# mistake even when Debian's version looks fine: it would create a second, unresolved
# source for the same import, and the two would disagree the first time either moved.
# docs/adr/0002-python-dependencies-from-pinned-wheels.md says why the split is here.
#
# Regenerate with: sh build/gather-apt-versions.sh

# @group appliance
# Everything below this marker is installed into the rootfs.

# The base layout and the shell. Debian minbase gives /bin/sh via dash.
#
# `mount` IS ITS OWN PACKAGE, and its absence is what stopped the first hardware boot dead
# at PID 1 line 62 with `mount: not found`. util-linux does NOT ship /usr/bin/mount; the
# package named `mount` does, and docs/boot-pipeline.md asserted the opposite in three
# places until that boot. util-linux stays named here because it is Essential and arrives
# whether or not it is pinned, and because the purge stage deletes the `agetty` it ships —
# a claim that needs its version in the one pinned list.
#
# THERE IS NO busybox. It was pinned for a `poweroff` that build/init never called: power
# off is reboot(2) with RB_POWER_OFF in aobs/adapters/real/power.py, and Debian's busybox
# ships /usr/bin/busybox with no applet symlinks, so the `poweroff` named in that comment
# would not have resolved either. Nothing in the repository invoked it.
dash=0.5.12-12
mount=2.41.5-0+deb13u1
util-linux=2.41.5-0+deb13u1

# `modprobe`, for step 3 of build/init, and the same kmod that build/mkiso.sh runs `depmod`
# from on the build host — so the modules.dep written into the image and the tool that reads
# it at boot are one implementation. Absent until the first hardware boot, where it made
# every module in the allowlist report `no <module> on this machine`: no camera and no USB
# HID driver, and not one failed step to say so, because a missing module is deliberately
# not a reason to refuse to boot.
kmod=34.2-2

# The interpreter. build/verify.py asserts both this and /bin/sh are present in the
# rootfs: the predecessor's first ISO had neither, and build/init could not have run a
# single line of itself.
python3=3.13.5-1

# Every EC operation on the appliance — derivation, ECDSA, Schnorr, address proof — runs
# here and never in embit's pure-Python fallback.
#
# VERIFIED against this exact .deb, not assumed: the shipped
# usr/lib/x86_64-linux-gnu/libsecp256k1.so.2 exports secp256k1_schnorrsig_sign32 and
# secp256k1_keypair_create (so BIP86's modules were enabled) AND the long-deprecated
# secp256k1_ec_privkey_negate alias, which the vendored embit's ctypes loader binds
# unconditionally and which upstream removed by 0.7. Both halves matter, and the second is
# the one that will break first: a future Debian shipping 0.7+ makes embit's loader raise,
# and secp256k1.py's bare `except:` turns that into a silent pure-Python appliance. That is
# what the build-time assertion in build/verify.py is for. Do not bump this pin without
# re-running it.
libsecp256k1-2=0.5.0-2+b1

# THE C++ RUNTIME THE WHEELS LINK, and the second pin in this file that exists because apt cannot
# see what the Python layer needs. `pip --target` unpacks the wheels from OUTSIDE the chroot, so no
# installed .deb declares a dependency on anything they link, and apt resolves a closure that is
# correct for the packages and blind to the application. Three objects in the image need this one:
# `zxingcpp.abi3.so`, `PIL/_avif...so` and pillow's bundled `libavif`.
#
# Measured: without it the second hardware boot reached `exec python3 -m aobs` and died at
# `import zxingcpp` — `libstdc++.so.6: cannot open shared object file`. The .deb was already in
# build/inputs/deb/appliance, fetched as a dependency of `apt`, which is not installed. Being in
# the pool is not being in the image, and only the pin puts it there.
libstdc++6=14.2.0-19

# `loadkeys` and the maps it loads. The Keymap adapter shells out to the first and reads
# the second. console-data is what makes layouts() return more than the default, and the
# full set ships deliberately: a user on AZERTY or ABNT2 typing a BIP39 passphrase
# through a US map creates a wallet they can never reopen, with no error and no signal.
kbd=2.7.1-2
console-data=2:1.12-9

# The kernel, unmodified, from trixie-security. build/mkiso.sh then prunes the modules
# tree to the allowlist and deletes the rest, kernel/net and drivers/net included.
linux-image-amd64=6.12.107-1

# @group harness
# The test tier only. The ISO carries no test runner and no package manager, and
# build/verify.py fails the build if any name below reaches the rootfs.

# HARNESS GROUP, NEVER THE APPLIANCE GROUP. verify.py fails the build if either name
# reaches the rootfs, and it matters twice over here: an appliance carrying git would
# have a network utility in the image. Both are needed by real tests rather than mocks —
# tests/test_verify_release.py drives the real verify-release.sh against a fixture
# release signed with scratch keys in a throwaway GNUPGHOME, and a fake gpg would be a
# test of our own mock. The one artifact aimed at people who trust nobody is the last one
# that should be tested against a stand-in.
git=1:2.47.3-0+deb13u1
gnupg=2.4.7-21+deb13u1

# Installs build/wheel-versions.txt, --no-index, from pre-fetched wheels only.
#
# It sits in the HARNESS group even though the image build uses it, and that is deliberate
# rather than sloppy: the two groups answer "may this survive into the shipped rootfs?",
# and for pip the answer is no. build/mkiso.sh installs it transiently, installs the
# wheels with it, then removes it, and build/verify.py fails the build if any package
# manager is present in the image it is about to pack. A third `@group buildtime` would
# have made the parser contract answer a different question than the one that matters.
python3-pip=25.1.1+dfsg-1

mmdebstrap=1.5.7-1+deb13u1
xorriso=1.5.6-1.2+b1
isolinux=3:6.04~git20190206.bf6db5b4+dfsg1-3.1
grub-efi-amd64-bin=2.12-9+deb13u2
